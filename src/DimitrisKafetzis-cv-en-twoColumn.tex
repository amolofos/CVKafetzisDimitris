% Widths below are derived from the real paracol columns (\textwidth and
% \columnsep), not from \paperwidth, so the sidebar box and the main-column
% text actually fit inside the columns paracol gives them. paracol splits
% (\textwidth-\columnsep) by the ratio, i.e. col1 = r(\textwidth-\columnsep);
% measured against the actual built PDF (col1 linewidth=159.2pt, col2
% linewidth=354.4pt at \columnratio{0.31}), confirming this formula -- the
% previous "0.31\textwidth-0.5\columnsep" version underestimated col1 by
% ~5pt and, through the subtraction below, overestimated col2 by the same
% ~5pt, which pushed the main-column text past paracol's real right edge
% and right up against the page margin. Keep the 0.31 here in sync with the
% \columnratio{} call further down.
\newlength\cvColOneWidth

\setlength\cvColTwoWidth{\textwidth-\columnsep-\cvColOneWidth}

% (inner xsep, overridden to 6pt below) and border.

% Width for sidebar text blocks that sit outside the box (summary, education,
% pie chart) -- these can use the full column width.
\setlength\cvSideWidthText{\cvColOneWidth-2pt}
% Width of the work-experience text in the main column. 15pt short of the
% column's real right edge, matching the rightmargin=15pt already used by
% the bullet lists elsewhere in this document, so the text keeps a visible
% margin before the page's right edge instead of running flush to it.
\setlength\cvEventTextWidth{\cvColTwoWidth-15pt}

% The box padding is narrower here than the class default (10pt), to leave
% more room for the box content in the narrower sidebar.

% The contact rows have an icon column to the left of the text column, so the
% text column must be narrower than the full box content width. Measured
% empirically: at "-16pt" the icon column + its 5pt column-sep actually need

	minimum height=\baselineskip,

	text width=\cvSideWidth-19pt,
